#### Fix unstable formatting of method chains with expandable object arguments (#20307 by @smg99)

Prettier now prefers expanding the last argument before splitting a method chain. This keeps formatting idempotent when an object argument needs multiple lines (for example at `--print-width 100`).

<!-- prettier-ignore -->
```jsx
// Input
async function a(harness, firstPatch) {
  harness.deps.patchDraft.mockReturnValueOnce(firstPatch).mockResolvedValueOnce({ ok: true, status: 200, body: { revision: 20, changed: ['span_a'], plan_name: 'orm' } });
}

// Prettier stable
async function a(harness, firstPatch) {
  harness.deps.patchDraft
    .mockReturnValueOnce(firstPatch)
    .mockResolvedValueOnce({
      ok: true,
      status: 200,
      body: { revision: 20, changed: ["span_a"], plan_name: "orm" },
    });
}

// Prettier main
async function a(harness, firstPatch) {
  harness.deps.patchDraft.mockReturnValueOnce(firstPatch).mockResolvedValueOnce({
    ok: true,
    status: 200,
    body: { revision: 20, changed: ["span_a"], plan_name: "orm" },
  });
}
```
