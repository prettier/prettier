#### Fix embedded template literal idempotency (#19725 by @fisker)

<!-- prettier-ignore -->
```jsx
// Input
const foo = html`
            <div>${getText({
            value,
            })}</div>
`;

// Prettier stable (first format)
const foo = html`
  <div>
    ${getText({
              value,
            })}
  </div>
`;

// Prettier stable (second format)
const foo = html`
  <div>
    ${getText({
      value,
    })}
  </div>
`;

// Prettier main
const foo = html`
  <div>
    ${getText({
      value,
    })}
  </div>
`;
```
