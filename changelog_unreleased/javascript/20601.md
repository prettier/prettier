#### Make unary expressions with leading comments in nested arguments idempotent (#20601 by @sanketpatel32)

<!-- prettier-ignore -->
```jsx
// Input
function t(window) {
  delete (/** @type {any} */ window).__pwned;
}

// Prettier stable
function t(window) {
  delete /** @type {any} */ window.__pwned;
}

// Prettier main
function t(window) {
  delete (/** @type {any} */ window.__pwned);
}
```
