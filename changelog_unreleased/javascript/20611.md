#### Fix unstable member chain formatting with `objectWrap: "preserve"` (#20611 by @Mohit-Gaur75)

Applies to JavaScript, TypeScript, and Flow.

<!-- prettier-ignore -->
```jsx
// Input
for ([] of []) {
  a, f3.get(t).set(l, { scope: t, methodName: l, endpointDefaults: d, decorations: a });
}

// Prettier stable (1st format)
for ([] of []) {
  (a,
    f3
      .get(t)
      .set(l, {
        scope: t,
        methodName: l,
        endpointDefaults: d,
        decorations: a,
      }));
}

// Prettier stable (2nd format)
for ([] of []) {
  (a,
    f3.get(t).set(l, {
      scope: t,
      methodName: l,
      endpointDefaults: d,
      decorations: a,
    }));
}

// Prettier main
for ([] of []) {
  (a,
    f3
      .get(t)
      .set(l, {
        scope: t,
        methodName: l,
        endpointDefaults: d,
        decorations: a,
      }));
}
```
