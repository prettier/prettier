#### Keep Flow return type comments after function parameters (#20222 by @mikamikasuki)

<!-- prettier-ignore -->
```jsx
// Input
async function doStuff(
  arg1 /*: string */, // The first arg
  arg2 /*: string */ // The second arg
) /*: Promise<string> */ // The return value
{
  console.log('thanks!');
}

// Prettier stable
async function doStuff(
  arg1 /*: string */, // The first arg
  arg2 /*: string */, // The second arg
  /*: Promise<string> */
) {
  // The return value
  console.log("thanks!");
}

// Prettier main
async function doStuff(
  arg1 /*: string */, // The first arg
  arg2 /*: string */ // The second arg
) /*: Promise<string> */ // The return value
{
  console.log("thanks!");
}
```
