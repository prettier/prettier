#### Preserve Angular syntax inside elements with `ngNonBindable` directive (#20066 by @splincode)

<!-- prettier-ignore -->
```html
<!-- Input -->
<span ngNonBindable>@if(foo){bar}</span>
<span ngNonBindable>@let foo=bar+baz;</span>

<!-- Prettier stable -->
<span ngNonBindable>
  @if (foo) {
    bar
  }
</span>
<span ngNonBindable>@let foo = bar + baz;</span>

<!-- Prettier main -->
<span ngNonBindable>@if(foo){bar}</span>
<span ngNonBindable>@let foo=bar+baz;</span>
```
