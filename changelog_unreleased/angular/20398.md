#### Format Storybook Angular CSF render templates (#20398 by @splincode)

Prettier now formats inline Angular templates returned by Storybook Angular CSF `render` arrow functions, using the existing Angular formatter. This supports both expression-bodied and block-bodied `render` functions while leaving unrelated `template` properties unchanged.

<!-- prettier-ignore -->
```ts
// Input
import type { StoryObj } from "@storybook/angular";

export const Default: StoryObj = {
  render: (args) => ({
    props: args,
    template: `<div>@if(visible){<span>{{name}}</span>}</div>`,
  }),
};

// Prettier stable
import type { StoryObj } from "@storybook/angular";

export const Default: StoryObj = {
  render: (args) => ({
    props: args,
    template: `<div>@if(visible){<span>{{name}}</span>}</div>`,
  }),
};

// Prettier main
import type { StoryObj } from "@storybook/angular";

export const Default: StoryObj = {
  render: (args) => ({
    props: args,
    template: `<div>
      @if (visible) {
        <span>{{ name }}</span>
      }
    </div>`,
  }),
};
```
