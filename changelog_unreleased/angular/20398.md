#### Format Storybook Angular CSF render templates (#20398 by @splincode)

Prettier now formats inline Angular templates returned by `render` functions and methods in named Storybook Angular CSF exports.

<!-- prettier-ignore -->
```ts
// Input
import type { StoryObj } from "@storybook/angular";

export const Default: StoryObj = {
  render: (args) => ({
    props: args,
    template: `<div>@if(visible){<span>{{name}}</span>}</div>`,
  }),
};

// Prettier stable
import type { StoryObj } from "@storybook/angular";

export const Default: StoryObj = {
  render: (args) => ({
    props: args,
    template: `<div>@if(visible){<span>{{name}}</span>}</div>`,
  }),
};

// Prettier main
import type { StoryObj } from "@storybook/angular";

export const Default: StoryObj = {
  render: (args) => ({
    props: args,
    template: `<div>
      @if (visible) {
        <span>{{ name }}</span>
      }
    </div>`,
  }),
};
```
