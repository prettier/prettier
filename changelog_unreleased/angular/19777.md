#### Add ng-model-options to recognized AngularJS attributes (#19777 by @xemlock)

<!-- prettier-ignore -->
```html
<!-- Input -->
<input ng-model="$ctrl.name" ng-model-options="{ debounce : 1000 , 'updateOn' : 'blur' }" />

<!-- Prettier stable -->
<input
  ng-model="$ctrl.name"
  ng-model-options="{ debounce : 1000 , 'updateOn' : 'blur' }"
/>

<!-- Prettier main -->
<input
  ng-model="$ctrl.name"
  ng-model-options="{ debounce: 1000, updateOn: 'blur' }"
/>
```
