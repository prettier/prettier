#### Preserve semicolon after bodyless declarations before block in `--no-semi` (#20638 by @itisrmk)

<!-- prettier-ignore -->
```ts
// Input
declare module "autoprefixer";

{
  alert();
}

declare function a();

{
  alert();
}

// Prettier stable (--no-semi)
declare module "autoprefixer"

{
  alert()
}

declare function a()

{
  alert()
}

// Prettier main (--no-semi)
declare module "autoprefixer";

{
  alert()
}

declare function a();

{
  alert()
}
```
