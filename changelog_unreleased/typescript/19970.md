#### Keep the parameter list stable when the return type holds a multi-line object type (#19970 by @lazerg)

Formatting such a function twice used to give two different results, because the
line break Prettier added inside the object type was read back as an
`objectWrap: "preserve"` break on the second run.

<!-- prettier-ignore -->
```tsx
// Input
function foo(
  row: ConnectionRow,
): Promise<{
  row: ConnectionRow;
  bundle: CredentialBundle;
  accessToken: string;
}> {
  return null as never;
}

// Prettier stable
function foo(row: ConnectionRow): Promise<{
  row: ConnectionRow;
  bundle: CredentialBundle;
  accessToken: string;
}> {
  return null as never;
}

// Prettier main
function foo(
  row: ConnectionRow,
): Promise<{
  row: ConnectionRow;
  bundle: CredentialBundle;
  accessToken: string;
}> {
  return null as never;
}
```
