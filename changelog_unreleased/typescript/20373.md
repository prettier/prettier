#### Indent type assertions after broken member chains (#20373 by @momooo15)

Keep the type following `as` or `satisfies` aligned with the last call in a broken member chain. Chains that fit on one line retain their existing indentation.

<!-- prettier-ignore -->
```ts
// Input
const result = [1, 2, 3]
  .map((n) => ({
    id: n,
  }))
  .filter((n) => n.id > 1) as {
  id: string;
};

// Prettier before this change (7664e36)
const result = [1, 2, 3]
  .map((n) => ({
    id: n,
  }))
  .filter((n) => n.id > 1) as {
  id: string;
};

// Prettier with this change
const result = [1, 2, 3]
  .map((n) => ({
    id: n,
  }))
  .filter((n) => n.id > 1) as {
    id: string;
  };
```
