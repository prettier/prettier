#### Preserve parentheses around `TSInstantiationExpression` before `!` and in `export default` (#20167)

<!-- prettier-ignore -->
```tsx
// Input
const a = (make<T>)!;
export default (class {})<string>;

// Prettier stable
const a = make<T>!;
export default class {}<string>;

// Prettier main
const a = (make<T>)!;
export default (class {}<string>);
```
