#### Fix misaligned jsdoc comment before a `satisfies` or `as` type (#19939 by @Kjubikstronk)

<!-- prettier-ignore -->
```tsx
// Input
const foo = {
  bar: 1,
} as
  /**
   * comment line 1
   * comment line 2
   */
  Foo;

// Prettier stable
const foo = {
  bar: 1,
} as /**
 * comment line 1
 * comment line 2
 */
Foo;

// Prettier main
const foo = {
  bar: 1,
} as
  /**
   * comment line 1
   * comment line 2
   */
  Foo;
```
