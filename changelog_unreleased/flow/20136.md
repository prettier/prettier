#### Support new Flow syntax (#20136 by @gkz)

Prettier now supports Flow syntax added in `flow-parser` 0.332.0, including template literal and constructor types, mapped-type modifiers and key remapping, abstract and override class members, inline import types, CommonJS import/export bindings, and `unique symbol` types.

<!-- prettier-ignore -->
```flow
// Input
abstract class Example { abstract value:string; }
type Constructor = new <T>(value:T)=>Container<T>;
type Getters<T> = {-readonly [K in keyof T as `get${K}`]-?: T[K]};
type Imported = import("package-name").Exported;
type Unique = unique symbol;
import Value=require("value-module");
export = Value;

// Prettier stable
// Not supported

// Prettier main
abstract class Example {
  abstract value: string;
}
type Constructor = new <T>(value: T) => Container<T>;
type Getters<T> = { -readonly [K in keyof T as `get${K}`]-?: T[K] };
type Imported = import("package-name").Exported;
type Unique = unique symbol;
import Value = require("value-module");
export = Value;
```
